_SYMBOLS = set(";:,.\\-\"'/()[]¿?¡!{}~<>|«»-—’\t\n\r")

_STOPWORDS = stopwords.words("spanish")
def load_file(path):
    with open(path, encoding="utf-8") as f:
        return f.read()

def normalize_text(text,
                   punct=False,
                   accents=False,
                   max_dup=2):
    """
        Estandariza el texto, remueve símbolos duplicados, acentos y puntuación
    """
    text = unicodedata.normalize('NFKC', text)
    text = text.lower()
    acentos = {'á':'a', 'é':'e', 'í':'i', 'ó':'o', 'ú':'u', 'ñ':'n'}
    n_str = []
    c_prev = ''
    cc_prev = 0
    for c in text:
        if not punct:
            if c in _SYMBOLS:
                continue
        if not accents:
            if c in acentos:
                c = acentos[c]
        if c_prev == c:
            cc_prev += 1
            if cc_prev >= max_dup:
                continue
        else:
            cc_prev = 0
        n_str.append(c)
        c_prev = c
    return "".join(n_str)

def ngrams(palabras, n, sep=" "):
    """
    Obtiene una lista de ngramas basado en la lista de palabras
    También funciona para lista de caracteres
    """
    output = []
    l = len(palabras)
    if (l - n + 1) <= 0:
        output.append(sep.join(palabras))
    for start in range(l - n + 1):
        t = sep.join(palabras[start:start+n])
        output.append(t)
    return output

def filter_words(text, remove_stopwords=True):
    """
    Filtra las palabras que inician http, "@" o "#", además de las stopwords y los números
    """
    wlist = text.split()
    o_text = []
    # filtrado de palabras
    for t in wlist:
        if t.startswith("http") or t.startswith("@") or t.startswith("#"):
            continue
        if remove_stopwords and t in _NORM_STOPWORDS:
            continue
        if t.isnumeric():
            continue
        o_text.append(t)
    return " ".join(o_text)

def preprocesa(text, remove_stopwords=True):
    text = normalize_text(text)
    text = filter_words(text, remove_stopwords=remove_stopwords)
    return text

_NORM_STOPWORDS = [normalize_text(sw)  for sw in _STOPWORDS]

def FrequencyDist_char(text):
    """
    Cuenta las frecuencias de caracteres en el texto
    """
    chars = {}
    text = text.split()

    # contar unigrams
    for w in text:
        for c in w:
            if c not in chars:
                chars[c] = 1
            else:
                chars[c] += 1
    return chars

def FrequencyDist_word(text):
    """
    Cuenta las frecuencias de palabras en el texto
    """
    words = {}
    text = text.split()

    # contar unigrams
    for w in text:
        if w not in words:
            words[w] = 1
        else:
            words[w] += 1

    return words
def entropy(frecuency):
    """
    Calcula la entropía dada las frecuencias
    """
    N = sum(frecuency.values())
    H = 0
    for c in frecuency:
        p = frecuency[c] / N
        H += p * math.log2(p)

    return -H
